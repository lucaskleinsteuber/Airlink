cmake_minimum_required(VERSION 3.20)
project(AirLink LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
get_filename_component(AIRLINK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
include(GNUInstallDirs)
include(CTest)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" AIRLINK_VERSION LIMIT_COUNT 1)
configure_file(src/version.h.in generated/version.h @ONLY)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network Widgets DBus Svg)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(airlink_core src/core.cpp src/core.h src/daemon.cpp src/daemon.h src/updates.cpp src/updates.h)
target_include_directories(airlink_core PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(airlink_core PUBLIC Qt6::Core Qt6::Network OpenSSL::Crypto)
target_compile_options(airlink_core PRIVATE -Wall -Wextra -Wpedantic -Werror -ffile-prefix-map=${AIRLINK_ROOT}=/AirLink -fdebug-prefix-map=${AIRLINK_ROOT}=/AirLink)
option(AIRLINK_BUILD_DESKTOP "Compile the desktop application" ON)
if(AIRLINK_BUILD_DESKTOP)
  add_executable(airlink src/main.cpp src/window.cpp src/window.h resources.qrc)
  target_link_libraries(airlink PRIVATE airlink_core Qt6::Widgets Qt6::DBus Qt6::Svg)
  target_compile_options(airlink PRIVATE -Wall -Wextra -Wpedantic -Werror -ffile-prefix-map=${AIRLINK_ROOT}=/AirLink -fdebug-prefix-map=${AIRLINK_ROOT}=/AirLink)
  install(TARGETS airlink RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(AIRLINK_HELPER)
  install(PROGRAMS "${AIRLINK_HELPER}" DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/airlink RENAME airlink-service)
endif()
install(FILES packaging/io.github.lucaskleinsteuber.AirLink.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES ../windows/AirLink.Windows/Assets/AirLink.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME io.github.lucaskleinsteuber.AirLink.svg)
install(FILES ../LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/airlink)
install(FILES packaging/io.github.lucaskleinsteuber.AirLink.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
if(BUILD_TESTING)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
  add_executable(airlink_core_tests tests/core_tests.cpp)
  target_link_libraries(airlink_core_tests PRIVATE airlink_core Qt6::Test)
  add_test(NAME core COMMAND airlink_core_tests)
  add_executable(airlink_update_verify tests/verify_release.cpp)
  target_link_libraries(airlink_update_verify PRIVATE airlink_core)
endif()
